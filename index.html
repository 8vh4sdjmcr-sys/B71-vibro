const NOMINAL_PERIOD = 3600000 / 14000;

// 257.142857 ms


const MIN_INTERVAL = 120;

const MAX_INTERVAL = 400;


const graph = document.getElementById("graph");

const ctx = graph.getContext("2d");


const startButton = document.getElementById("start");

const stopButton = document.getElementById("stop");

const resetButton = document.getElementById("reset");


const rateElement = document.getElementById("rate");

const beatElement = document.getElementById("beatError");

const signalElement = document.getElementById("signal");

const qualityElement = document.getElementById("quality");

const statusElement = document.getElementById("status");


let audioContext = null;

let analyser = null;

let microphone = null;

let mediaStream = null;


let running = false;

let animationFrame = null;


let audioData = null;


let envelope = 0;

let dc = 0;


let lastPeakTime = 0;


let peaks = [];


const MAX_PEAKS = 60;



/*

----------------------------------------------------------

Pomocné funkce

----------------------------------------------------------

*/


function median(values) {


    if (values.length === 0)

        return 0;


    const a = [...values].sort((a,b) => a-b);


    return a[Math.floor(a.length / 2)];

}



function setStatus(text) {


    statusElement.textContent = text;


}



/*

----------------------------------------------------------

Reset měření

----------------------------------------------------------

*/


function resetMeasurement() {


    peaks = [];


    lastPeakTime = 0;


    rateElement.textContent = "—";

    beatElement.textContent = "—";

    signalElement.textContent = "—";

    qualityElement.textContent = "—";


    drawGraph([]);


}



/*

----------------------------------------------------------

Výpočet výsledků

----------------------------------------------------------

*/


function calculateResults() {


    if (peaks.length < 5)

        return;


    const intervals = [];


    for (let i = 1; i < peaks.length; i++) {


        const d =

            peaks[i].time -

            peaks[i - 1].time;


        if (

            d >= MIN_INTERVAL &&

            d <= MAX_INTERVAL

        ) {


            intervals.push(d);


        }


    }


    if (intervals.length < 4)

        return;



    /*

    ------------------------------------------

    RATE

    ------------------------------------------

    */


    const period = median(intervals);


    const rate =

        ((period - NOMINAL_PERIOD)

        / NOMINAL_PERIOD)

        * 86400

        / 1000;


    rateElement.textContent =

        (rate >= 0 ? "+" : "") +

        rate.toFixed(1) +

        " s/d";



    /*

    ------------------------------------------

    BEAT ERROR

    ------------------------------------------


    Rozdělíme intervaly na dvě skupiny:


    TIK → TAK

    TAK → TIK


    Rozdíl jejich průměrných hodnot

    odpovídá beat error.

    */


    const a = [];

    const b = [];


    for (let i = 0; i < intervals.length; i++) {


        if (i % 2 === 0)

            a.push(intervals[i]);

        else

            b.push(intervals[i]);


    }


    if (a.length && b.length) {


        const beatError =

            Math.abs(

                median(a) -

                median(b)

            );


        beatElement.textContent =

            beatError.toFixed(1) +

            " ms";


    }



    /*

    ------------------------------------------

    SIGNAL

    ------------------------------------------

    */


    const recentPeaks =

        peaks.slice(-12);


    if (recentPeaks.length) {


        const levels =

            recentPeaks.map(

                p => p.level

            );


        const level =

            median(levels);


        signalElement.textContent =

            Math.round(level * 100) +

            " %";


    }



    /*

    ------------------------------------------

    QUALITY

    ------------------------------------------

    */


    let good = 0;


    for (const d of intervals) {


        if (

            Math.abs(

                d - NOMINAL_PERIOD

            ) < 30

        ) {


            good++;


        }


    }


    const quality =

        Math.round(

            100 *

            good /

            intervals.length

        );


    qualityElement.textContent =

        quality + " %";



    /*

    ------------------------------------------

    Stav detekce

    ------------------------------------------

    */


    if (quality > 80) {


        setStatus(

            "Signál OK — pravidelné tikání detekováno."

        );


    }

    else {


        setStatus(

            "Naslouchám… hledám pravidelný signál."

        );


    }


}



/*

----------------------------------------------------------

Detekce špiček

----------------------------------------------------------

*/


function detectPeak(samples) {


    const now =

        performance.now();



    /*

    Automatický práh.


    Hluk držíme nízko.

    */


    const recentLevels =

        samples.slice(-200);


    const noise =

        median(recentLevels);



    const threshold =

        Math.max(

            0.025,

            noise * 2.8

        );



    /*

    Hledání lokálního maxima.

    */


    for (

        let i = 2;

        i < samples.length - 2;

        i++

    ) {


        const x = samples[i];


        if (x < threshold)

            continue;


        if (x < samples[i-1])

            continue;


        if (x < samples[i+1])

            continue;


        /*

        Nebereme několik špiček

        jednoho tiknutí.

        */


        if (

            now - lastPeakTime <

            MIN_INTERVAL

        ) {


            continue;


        }



        /*

        --------------------------------------

        Kandidát na tik/tak

        --------------------------------------

        */


        lastPeakTime = now;


        const peak = {


            time: now,


            level:

                Math.min(

                    1,

                    x * 3

                )


        };



        peaks.push(peak);



        if (

            peaks.length >

            MAX_PEAKS

        ) {


            peaks.shift();


        }



        calculateResults();


        break;


    }


}



/*

----------------------------------------------------------

Zpracování mikrofonu

----------------------------------------------------------

*/


function processAudio() {


    if (!running)

        return;



    analyser.getFloatTimeDomainData(

        audioData

    );



    const samples = [];



    for (

        let i = 0;

        i < audioData.length;

        i++

    ) {


        const raw =

            audioData[i];



        /*

        --------------------------------------

        Odstranění pomalé DC složky

        --------------------------------------

        */


        dc =

            dc * 0.995 +

            raw * 0.005;



        const highPass =

            raw - dc;



        /*

        --------------------------------------

        Absolutní hodnota

        --------------------------------------

        */


        const rectified =

            Math.abs(highPass);



        /*

        --------------------------------------

        Obálka signálu

        --------------------------------------

        */


        envelope =

            envelope * 0.88 +

            rectified * 0.12;



        /*

        Zesílení pouze pro zobrazení

        */


        const display =

            Math.min(

                1,

                envelope * 8

            );



        samples.push(display);


    }



    detectPeak(samples);


    drawGraph(samples);



    animationFrame =

        requestAnimationFrame(

            processAudio

        );


}



/*

----------------------------------------------------------

Graf

----------------------------------------------------------

*/


function drawGraph(samples) {


    const rect =

        graph.getBoundingClientRect();


    const dpr =

        window.devicePixelRatio || 1;



    if (

        graph.width !==

        rect.width * dpr

    ) {


        graph.width =

            rect.width * dpr;


        graph.height =

            rect.height * dpr;


        ctx.scale(dpr,dpr);


    }



    const width =

        rect.width;


    const height =

        rect.height;



    /*

    Pozadí

    */


    ctx.fillStyle =

        "#050505";


    ctx.fillRect(

        0,

        0,

        width,

        height

    );



    /*

    Mřížka

    */


    ctx.strokeStyle =

        "#222";


    ctx.lineWidth = 1;



    for (

        let y = 1;

        y < 5;

        y++

    ) {


        const yy =

            y * height / 5;


        ctx.beginPath();


        ctx.moveTo(

            0,

            yy

        );


        ctx.lineTo(

            width,

            yy

        );


        ctx.stroke();


    }



    /*

    Obálka zvuku

    */


    if (samples.length > 1) {


        ctx.strokeStyle =

            "#ddd";


        ctx.lineWidth = 1.5;


        ctx.beginPath();



        for (

            let i = 0;

            i < samples.length;

            i++

        ) {


            const x =

                i *

                width /

                samples.length;



            const y =

                height -

                10 -

                samples[i] *

                (height - 20);



            if (i === 0)

                ctx.moveTo(x,y);

            else

                ctx.lineTo(x,y);


        }


        ctx.stroke();


    }



    /*

    Detekované impulzy

    */


    const now =

        performance.now();


    const WINDOW = 6000;



    for (

        const p of peaks

    ) {


        const age =

            now - p.time;



        if (

            age < 0 ||

            age > WINDOW

        )

            continue;



        const x =

            width -

            age /

            WINDOW *

            width;



        const y =

            height -

            10 -

            p.level *

            (height - 20);



        ctx.fillStyle =

            "#fff";



        ctx.beginPath();


        ctx.arc(

            x,

            y,

            4,

            0,

            Math.PI * 2

        );


        ctx.fill();


    }



    /*

    Cílový interval

    */


    ctx.fillStyle =

        "#777";


    ctx.font =

        "12px Arial";


    ctx.fillText(

        "14 000 A/h · 257,14 ms",

        10,

        18

    );


}

startButton.onclick =

async function() {


    try {


        /*

        Povolení mikrofonu.

        */


        mediaStream =

            await navigator

            .mediaDevices

            .getUserMedia({


                audio: {


                    echoCancellation: false,


                    noiseSuppression: false,


                    autoGainControl: false


                }


            });



        /*

        AudioContext

        */


        audioContext =

            new (

                window.AudioContext ||

                window.webkitAudioContext

            )();



        /*

        Mikrofon

        */


        microphone =

            audioContext

            .createMediaStreamSource(

                mediaStream

            );



        /*

        Analyser

        */


        analyser =

            audioContext

            .createAnalyser();



        analyser.fftSize =

            1024;



        analyser.smoothingTimeConstant =

            0;



        microphone.connect(

            analyser

        );



        audioData =

            new Float32Array(

                analyser.fftSize

            );



        /*

        Reset

        */


        resetMeasurement();



        envelope = 0;

        dc = 0;

        lastPeakTime = 0;



        running = true;



        startButton.disabled =

            true;


        stopButton.disabled =

            false;



        setStatus(

            "Mikrofon běží. Přilož telefon k hodinám a čekej na tikání…"

        );



        processAudio();



    }

    catch(error) {


        setStatus(

            "Nelze spustit mikrofon: " +

            error.message

        );


    }


};
